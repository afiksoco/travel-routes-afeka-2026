זרימת הרשאות (JWT)


  • הרשמה/התחברות → Express מנפיק Access Token (24 שעות) + Refresh Token (7 ימים)
  • סיסמאות מוצפנות עם bcrypt + salt (12 rounds)
  • JWT מכיל: { sub: userId, name: studentName }
  • Next.js Middleware מאמת טוקן בכל כניסה לדף מוגן
  • רענון שקוף (Silent Refresh) — המשתמש לא מרגיש
  • Refresh Token מסובב (rotated) בכל רענון — אבטחה מוגברת

דגימת קוד — JWT Middleware

// middleware.ts (Next.js)
export function middleware(request: NextRequest) {
  const token = request.cookies.get("access_token")?.value;
  if (!token) {
    return NextResponse.redirect(new URL("/login", request.url));
  }
  // Decode JWT, check expiry
  const payload = JSON.parse(atob(token.split(".")[1]));
  if (payload.exp * 1000 < Date.now()) {
    // Redirect to login for re-auth
  }
}
